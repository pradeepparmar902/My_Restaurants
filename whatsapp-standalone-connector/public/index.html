// Send message from any frontend or React / Vue / HTML app:
async function sendWhatsApp(phoneNumber, messageText) {
  const response = await fetch('http://localhost:5050/api/send', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      to: phoneNumber,
      text: messageText
    })
  });
  const data = await response.json();
  if (data.ok) {
    alert('WhatsApp Message Sent!');
  } else {
    alert('Error: ' + data.error);
  }
}
